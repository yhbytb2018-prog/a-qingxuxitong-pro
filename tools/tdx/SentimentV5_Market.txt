{Sentiment V5.0 Market 大盘专用｜上证指数副图，带ADL涨跌家数}
{五大因子权重：情绪定位0.25 盈亏效应0.25 广度量能0.20 题材结构0.20 主线板块结构0.10}
{风险五档：≥80 过热警示 / 65~79 低风险 / 45~64 中等 / 25~44 高风险 / <25 极高风险}
{重要：ADVANCE/DECLINE 是大盘指数专用函数，必须挂在指数K线副图（如上证指数 999999），
 个股K线下无涨跌家数数据，请用 SentimentV5_Lite 个股版}
{对外部供稿版的修正：LIMIT→MAX(MIN)、CONST 误用移除、MOM/VOL_RATIO 去中性锚、
 ADL_SCORE 重构（原稿 +15 固定漂移且 ADL 实际权重仅 0.3）、风险线对齐 V5 五档 25/45/65/80}

{==== 参数区（可在公式参数面板调整）====}
N1:=13;
N2:=26;
N3:=5;

{==== 1.情绪定位因子(0~100)：RSI + 5日动量合成，50为中性锚 ====}
LC:=REF(CLOSE,1);
RSI_VAL:=SMA(MAX(CLOSE-LC,0),N1,1)/SMA(ABS(CLOSE-LC),N1,1)*100;
MOM:=MA(CLOSE,N3)/REF(MA(CLOSE,N3),N3)*50;
F1:=MAX(MIN(50+(RSI_VAL-50)*0.4+(MOM-50)*0.6,100),0);

{==== 2.盈亏效应因子(0~100)：振幅/长上影扣分 + 5日持有回报加分 ====}
AMP:=(HIGH-LOW)/REF(CLOSE,1)*100;
UPSHADOW:=(HIGH-MAX(CLOSE,OPEN))/REF(CLOSE,1)*100;
RET5:=HHV(CLOSE,5)/LLV(CLOSE,5);
F2:=MAX(MIN(100-AMP*2-UPSHADOW*1.5+(RET5-1)*100,100),0);

{==== 3.广度量能因子(0~100)：ADL涨跌家数为主力信号 + 量比 + 价格趋势 ====}
{当日上涨家数-下跌家数，5日平滑；沪市典型区间 ±1000，按 ±40 分封顶映射}
ADLD:=ADVANCE-DECLINE;
ADL_MA:=MA(ADLD,5);
ADL_SCORE:=MAX(MIN(ADL_MA/25,40),-40);
VOLR:=VOL/MA(VOL,N2);
PTREND:=(CLOSE-MA(CLOSE,N2))/MA(CLOSE,N2)*100;
F3:=MAX(MIN(50+ADL_SCORE*0.8+(VOLR-1)*10+PTREND*1.5,100),0);

{==== 4.题材结构因子(0~100)：MACD 动量扩散（大盘版代理） ====}
DIFF:=EMA(CLOSE,12)-EMA(CLOSE,26);
DEA:=EMA(DIFF,9);
MACDH:=(DIFF-DEA)/CLOSE*10000;              {柱体按点位百分比归一，典型 ±30}
F4:=MAX(MIN(50+MACDH*1.2,100),0);

{==== 5.主线板块内部结构因子(0~100)：MA5/MA20 短期趋势斜率 ====}
TREND:=(MA(CLOSE,5)/MA(CLOSE,20)-1)*100;    {典型 ±3%}
F5:=MAX(MIN(50+TREND*15,100),0);

{==== V5.0 加权综合得分 ====}
SCORE:=F1*0.25+F2*0.25+F3*0.20+F4*0.20+F5*0.10;

{==== 风险分界参考线（V5 五档，对齐 25/45/65/80）====}
零线:0,COLORGRAY,DOTLINE;
顶线:100,COLORGRAY,DOTLINE;
高分线:25,COLORRED,DOTLINE;
中分线:45,COLORYELLOW,DOTLINE;
低分线:65,COLORGREEN,DOTLINE;
过热线:80,COLORLIRED,DOTLINE;

{==== 因子分项曲线 ====}
F1P:F1,COLORBLUE;
F2P:F2,COLORCYAN;
F3P:F3,COLORYELLOW;
F4P:F4,COLORMAGENTA;
F5P:F5,COLORGRAY;

{==== 综合得分主线 + 分段变色（DRAWNULL 防零值污染）====}
SCORE_LINE:SCORE,COLORRED,LINETHICK2;
段过热:IF(SCORE>=80,SCORE,DRAWNULL),COLORLIRED,LINETHICK3;
段强势:IF(SCORE>=65 AND SCORE<80,SCORE,DRAWNULL),COLORRED,LINETHICK3;
段中性:IF(SCORE>=45 AND SCORE<65,SCORE,DRAWNULL),COLORYELLOW,LINETHICK3;
段弱势:IF(SCORE>=25 AND SCORE<45,SCORE,DRAWNULL),COLORGREEN,LINETHICK3;
段冰点:IF(SCORE<25,SCORE,DRAWNULL),COLORBLUE,LINETHICK3;

{==== 末栏标注 ====}
DRAWTEXT(ISLASTBAR,SCORE,' 综合分'),COLORRED;
DRAWTEXT(ISLASTBAR,80,'过热区'),COLORLIRED;
DRAWTEXT(ISLASTBAR,65,'低风险'),COLORGREEN;
DRAWTEXT(ISLASTBAR,45,'中等风险'),COLORYELLOW;
DRAWTEXT(ISLASTBAR,25,'高风险'),COLORRED;
